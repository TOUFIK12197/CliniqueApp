-- ============================================
-- SCRIPT DE CRÉATION - Clinique SantéPlus
-- ============================================

CREATE DATABASE IF NOT EXISTS clinique_santeplus 
CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;

USE clinique_santeplus;

-- Table Utilisateur
CREATE TABLE Utilisateur (
    idUtilisateur INT AUTO_INCREMENT PRIMARY KEY,
    login VARCHAR(50) NOT NULL UNIQUE,
    motDePasse VARCHAR(255) NOT NULL,
    role ENUM('ADMIN','MEDECIN','SECRETAIRE') NOT NULL,
    actif BOOLEAN DEFAULT TRUE,
    nomUtilisateur VARCHAR(100) NOT NULL,
    prenomUtilisateur VARCHAR(100) NOT NULL,
    telephoneUtilisateur VARCHAR(20),
    emailUtilisateur VARCHAR(150)
);

-- Table Specialite
CREATE TABLE specialite (
    idspecialite INT AUTO_INCREMENT PRIMARY KEY,
    libelle VARCHAR(100) NOT NULL,
    description TEXT
);

-- Table Statistiques
CREATE TABLE Statistiques (
    idstatistiques INT AUTO_INCREMENT PRIMARY KEY,
    periode VARCHAR(50),
    nbRendezVous INT DEFAULT 0,
    nbPatient INT DEFAULT 0,
    nbAnnulations INT DEFAULT 0,
    nbconsultation INT DEFAULT 0
);

-- Table Administrateur
CREATE TABLE Administrateur (
    idUtilisateur INT PRIMARY KEY,
    idspecialite INT,
    idstatistiques INT,
    FOREIGN KEY (idUtilisateur) REFERENCES Utilisateur(idUtilisateur) ON DELETE CASCADE,
    FOREIGN KEY (idspecialite) REFERENCES specialite(idspecialite),
    FOREIGN KEY (idstatistiques) REFERENCES Statistiques(idstatistiques)
);

-- Table Medecin
CREATE TABLE Medecin (
    idUtilisateur INT PRIMARY KEY,
    numOrdre VARCHAR(50) NOT NULL UNIQUE,
    FOREIGN KEY (idUtilisateur) REFERENCES Utilisateur(idUtilisateur) ON DELETE CASCADE
);

-- Table Patient
CREATE TABLE Patient (
    idPatient INT AUTO_INCREMENT PRIMARY KEY,
    nomPatient VARCHAR(100) NOT NULL,
    prenomPatient VARCHAR(100) NOT NULL,
    dateNaissance DATE,
    telephonePatient VARCHAR(20),
    emailPatient VARCHAR(150),
    adresse VARCHAR(255),
    groupeSanguin VARCHAR(5)
);

-- Table Secretaire
CREATE TABLE Secretaire (
    idUtilisateur INT PRIMARY KEY,
    idPatient INT,
    FOREIGN KEY (idUtilisateur) REFERENCES Utilisateur(idUtilisateur) ON DELETE CASCADE,
    FOREIGN KEY (idPatient) REFERENCES Patient(idPatient)
);

-- Table Consultation
CREATE TABLE Consultation (
    idConsultation INT AUTO_INCREMENT PRIMARY KEY,
    DateConsult DATE NOT NULL,
    diagnostic TEXT,
    tension VARCHAR(20),
    notes TEXT,
    prescription TEXT,
    poids DECIMAL(5,2)
);

-- Table RendezVous
CREATE TABLE RendezVous (
    idRdv INT AUTO_INCREMENT PRIMARY KEY,
    dateHeure DATETIME NOT NULL,
    motif VARCHAR(255),
    statut ENUM('EN_ATTENTE','CONFIRME','ANNULE','TERMINE') DEFAULT 'EN_ATTENTE',
    idConsultation INT,
    FOREIGN KEY (idConsultation) REFERENCES Consultation(idConsultation)
);

-- Table de relation : possede (Medecin - Specialite)
CREATE TABLE possede (
    idUtilisateur INT NOT NULL,
    idspecialite INT NOT NULL,
    PRIMARY KEY (idUtilisateur, idspecialite),
    FOREIGN KEY (idUtilisateur) REFERENCES Medecin(idUtilisateur),
    FOREIGN KEY (idspecialite) REFERENCES specialite(idspecialite)
);

-- Table de relation : prend (Medecin - Patient - RendezVous)
CREATE TABLE prend (
    idUtilisateur INT NOT NULL,
    idPatient INT NOT NULL,
    idRdv INT NOT NULL,
    PRIMARY KEY (idUtilisateur, idPatient, idRdv),
    FOREIGN KEY (idUtilisateur) REFERENCES Medecin(idUtilisateur),
    FOREIGN KEY (idPatient) REFERENCES Patient(idPatient),
    FOREIGN KEY (idRdv) REFERENCES RendezVous(idRdv)
);

-- Table de relation : realise (Medecin - Consultation)
CREATE TABLE realise (
    idUtilisateur INT NOT NULL,
    idConsultation INT NOT NULL,
    PRIMARY KEY (idUtilisateur, idConsultation),
    FOREIGN KEY (idUtilisateur) REFERENCES Medecin(idUtilisateur),
    FOREIGN KEY (idConsultation) REFERENCES Consultation(idConsultation)
);

-- Table de relation : concerne (Medecin - Patient - Consultation)
CREATE TABLE concerne (
    idUtilisateur INT NOT NULL,
    idPatient INT NOT NULL,
    idConsultation INT NOT NULL,
    PRIMARY KEY (idUtilisateur, idPatient, idConsultation),
    FOREIGN KEY (idUtilisateur) REFERENCES Medecin(idUtilisateur),
    FOREIGN KEY (idPatient) REFERENCES Patient(idPatient),
    FOREIGN KEY (idConsultation) REFERENCES Consultation(idConsultation)
);
